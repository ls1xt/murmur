2. **TrayIcon class interface:**
```python
class TrayIcon:
    def __init__(self, on_quit: Callable[[], None] | None = None) -> None:
        self._on_quit = on_quit
        self._icon: Any = None  # pystray.Icon (optional import)
        self._thread: threading.Thread | None = None
        self._state = "idle"
        self._available = False

    def start(self) -> None:
        """Start tray icon in background daemon thread. Non-blocking. Graceful on failure."""
        try:
            import pystray  # late import -- optional dependency
            from PIL import Image, ImageDraw
            # ... create icon with menu, start in daemon thread
            self._available = True
        except Exception as e:
            print(f"murmur: tray icon unavailable: {e}", file=sys.stderr)
            self._available = False

    def set_state(self, state: str) -> None:
        """Update tray icon to reflect new state. Thread-safe. No-op if tray unavailable."""
        if not self._available or self._icon is None:
            return
        self._state = state
        self._icon.icon = self._create_icon(state)
        self._icon.title = f"Murmur - {state.capitalize()}"
        self._icon.update_menu()

    def stop(self) -> None:
        """Stop tray icon. Safe to call even if not running."""
        if self._icon is not None:
            self._icon.stop()
```
