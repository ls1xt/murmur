Create `murmur/feedback/notify.py` with a `Notifier` class:

```python
class Notifier:
    def __init__(self, level: str = "errors") -> None:
        # level: "off", "errors", "verbose"
        self._level = level

    def error(self, message: str) -> None:
        """Send persistent error notification (urgency=critical). Only if level != 'off'."""
        if self._level == "off":
            return
        self._send(message, urgency="critical")

    def info(self, message: str) -> None:
        """Send auto-expiring informational notification. Only if level == 'verbose'."""
        if self._level != "verbose":
            return
        self._send(message, urgency="normal", expire_ms=5000)

    def _send(self, message: str, urgency: str = "normal", expire_ms: int | None = None) -> None:
        """Shell out to notify-send. Best-effort, never crashes."""
        cmd = ["notify-send", f"--urgency={urgency}", "Murmur", message]
        if expire_ms is not None:
            cmd.insert(-2, f"--expire-time={expire_ms}")
        with contextlib.suppress(FileNotFoundError, subprocess.TimeoutExpired):
            subprocess.run(cmd, timeout=5, capture_output=True)
```
