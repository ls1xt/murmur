Create `packaging/aur/` directory and write the PKGBUILD.

**AUR name decision (per CONTEXT.md Claude's discretion):** Use `python-murmur` — this is the Arch naming convention for Python packages and avoids a potential conflict with the `murmur` package (Mumble server) that may exist in AUR.

Read pyproject.toml for the current version before writing.

```bash
# Maintainer: OWNER <email@example.com>
pkgname=python-murmur
pkgver=0.1.0
pkgrel=1
pkgdesc="System-wide voice dictation for Linux and macOS"
arch=('x86_64')
url="https://github.com/OWNER/murmur"
license=('MIT')
depends=(
    'python>=3.10'
    'python-pillow'
    'python-pystray'
    'python-sounddevice'
    'portaudio'
)
makedepends=(
    'python-build'
    'python-installer'
    'python-wheel'
    'python-hatchling'
)
# Note: assemblyai and customtkinter are NOT in official Arch repos or AUR as python-assemblyai.
# They will be installed via pip as part of the package. See .install file.
optdepends=(
    'ydotool: text injection on Wayland (Linux)'
    'python-evdev: keyboard hotkey detection on Linux'
)
source=("https://files.pythonhosted.org/packages/source/m/murmur/murmur-${pkgver}.tar.gz")
sha256sums=('PLACEHOLDER_SHA256_AFTER_PYPI_PUBLISH')

build() {
    cd "murmur-${pkgver}"
    python -m build --wheel --no-isolation
}

package() {
    cd "murmur-${pkgver}"
    python -m installer --destdir="$pkgdir" dist/*.whl
    install -Dm644 LICENSE "${pkgdir}/usr/share/licenses/${pkgname}/LICENSE"
    # Install systemd service file
    install -Dm644 murmur.service "${pkgdir}/usr/lib/systemd/user/murmur.service"
}
```
